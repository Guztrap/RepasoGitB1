// CASO DE PRUEBA 3: Consolidación de Datos con Estructura Correcta
// ==========================================
PRUEBA_UNITARIA Test_ConsolidarDatos_FormatoCorrecto()
    // 1. Preparación (Arrange)
    materias_mock = ["Matemáticas", "Programación"]
    maestros_mock = ["Prof. García", "Ing. López"]
    calificaciones_mock = [95, 100]

    // 2. Ejecución (Act)
    resultado = ConsolidarDatos(materias_mock, maestros_mock, calificaciones_mock)

    // 3. Afirmación / Verificación (Assert)
    // Verifica el orden especifico [Materia, Maestro, Calificacion]
    AFIRMAR(LONGITUD(resultado) == 2)
    AFIRMAR(resultado[0][0] == "Matemáticas")   // Materia
    AFIRMAR(resultado[0][1] == "Prof. García")   // Maestro
    AFIRMAR(resultado[0][2] == 95)             // Calificación
FIN PRUEBA_UNITARIA
